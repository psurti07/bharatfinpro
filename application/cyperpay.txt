<?php

class PaymentController extends CI_Controller
{
    public function __construct()
    {
        parent::__construct();
        $this->load->library('session');
        $this->load->helper(['url', 'form']);
        $this->load->library('curl'); // Assuming you have a curl library
    }
	

    public function initiate_collect()
    {
        $request = [
            "method" => "POST",
            "url" => "payin/initiate-collect",
            "parameter" => [
                "sender_vpa" => "patel6466bimal@okhdfcbank",
                "sender_name" => "Bimal patel",
                "sender_mobile" => "+919924276751",
                "receiver_vpa" => "",
                "amount" => "10",
                "remarks" => "Initiate Collect",
                "refid" => rand(1000, 9999),
                "expiry" => "15"
            ]
        ];

        $response = $this->hit($request);
        $response = $this->final_response($response);
        log_message('info', 'final Response - ' . json_encode($response));
        echo json_encode($response);
    }

    public function status_enquiry()
    {
        $refid = $this->session->userdata('refid');
        $data = [
            "method" => "POST",
            "url" => "payin/status-enquiry",
            "parameter" => [
                "refid" => $refid
            ]
        ];

        $response1 = $this->hit($data);
        $response = $this->final_response($response1);
        $this->session->set_userdata('cipherResponse', $response);

        $payment_message = $this->cipher_payment_status($response['data']['status']);
        $url = $_SERVER['HTTP_REFERER'];
        $path = parse_url($url, PHP_URL_PATH);
        $segments = explode('/', trim($path, '/'));
        $second_last_segment = $segments[count($segments) - 2];

        $actual_return_url = $second_last_segment === 'self-apply' ? 'https://kreditbazar.com/self-apply/offer-3-response' : 'https://kreditbazar.com/loan-agent/offer-3-response';

        $redirect_url = $payment_message === 'Transaction Successfull' ? $actual_return_url : 'https://kreditbazar.com/cipher-payment-failed/';
        log_message('info', $redirect_url);
        echo json_encode(['type' => 'SUCCESS', 'message' => $payment_message, 'redirectUrl' => $redirect_url, 'data' => $response]);
    }

    public function dynamic_qr()
    {
        $refid = rand(1000, 9999);
        $request = [
            "method" => "POST",
            "url" => "payin/dynamic-qr",
            "parameter" => [
                'receiver_vpa' => "cpy.kreditbazar@finobank",
                'amount' => "299",
                'remarks' => "Dynamic QR",
                'refid' => $refid,
                'expiry' => "2",
                'type' => "QR"
            ]
        ];

        $this->session->set_userdata('refid', $refid);
        log_message('info', json_encode($request));
        $response = $this->hit($request);
        $response = $this->final_response($response);
        log_message('info', 'final Response - ' . json_encode($response));

        $this->load->view('pg/cipherQR', ['response' => $response]);
    }

    private function hit($req_data)
    {
        $url = base_url() . $req_data['url'];
        $jwt = $this->get_jwt_token();

        $parameter = !empty($req_data['parameter']) ? json_encode($req_data['parameter']) : "";
        $info = $this->final_request($parameter);

        $curl = curl_init();
        curl_setopt_array($curl, [
            CURLOPT_URL => $url,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_CUSTOMREQUEST => $req_data['method'],
            CURLOPT_POSTFIELDS => json_encode($info['payload']),
            CURLOPT_HTTPHEADER => [
                "Token: $jwt",
                "Auth: {$info['Auth']}",
                "Key: {$info['Key']}",
                "Content-Type: application/json",
                "User-Agent: PostmanRuntime/7.29.2"
            ],
        ]);

        $response = curl_exec($curl);
        if (curl_errno($curl)) {
            $resp = ["errorCode" => "PAYSPRINT-001", "error_code" => curl_errno($curl), "message" => curl_error($curl), "errorMessage" => "Unable to get response please try again later"];
        } else {
            $resp = $this->parse_response($response);
        }
        curl_close($curl);
        return $resp;
    }

    private function final_request($parameters = "")
    {
        $salt = bin2hex(openssl_random_pseudo_bytes(8));
        $data = $this->generate_aes_key($salt);
        $key = $data[0];
        $iv = $data[1];

        if ($parameters !== "") {
            $encrypted = openssl_encrypt(json_encode($parameters), 'aes-128-cbc', $key, OPENSSL_RAW_DATA, $iv);
            $encrypted = base64_encode($encrypted);
        }

        $encrypted_salt = $this->rsa_encrypt($salt);

        return [
            'Auth' => $encrypted_salt,
            'Key' => $encrypted_salt,
            'payload' => $parameters ? ['requestData' => $encrypted] : null
        ];
    }

    private function parse_response($response)
    {
        return json_decode($response, true);
    }

    private function final_response($response)
    {
        if (!isset($response['returnData'])) {
            return $response;
        }
        $response_data = $response['returnData'];
        $encrypted = base64_decode($response_data);
        $decrypted = openssl_decrypt($encrypted, 'aes-128-cbc', $this->session->userdata('aesKey'), OPENSSL_RAW_DATA, $this->session->userdata('aesIv'));
        return json_decode($decrypted, true);
    }

    private function generate_aes_key($salt)
    {
        $salt = hex2bin($salt);
        $passphrase = 'CipherPay API Payout';
        $iteration_count = 10000;
        $key_size = 128;
        $hash_algorithm = 'sha1';
        $key = openssl_pbkdf2($passphrase, $salt, $key_size / 8, $iteration_count, $hash_algorithm);
        $this->session->set_userdata('aesKey', $key);
        $this->session->set_userdata('aesIv', bin2hex($salt));
        return [$key, bin2hex($salt)];
    }

    private function rsa_encrypt($data)
    {
        $public_key = "YOUR_PUBLIC_KEY_HERE";
        openssl_public_encrypt($data, $encrypted, $public_key);
        return base64_encode($encrypted);
    }
}
